# 大模型调用 + 工具调用 + 条件路由 + ReAct + 短期记忆

在 LangGraph 里，用 `Checkpointer`（存档器）。每次 AI 或 人类 说一句话，系统就自动保存一下。这样模型在回答你下一句时，可以把上几句一起喂给它，保证对话连贯。

```python
# ==============================================================================
# 第一部分：导入所需的工具箱
# ==============================================================================
import os
from typing import Literal
from langchain_deepseek import ChatDeepSeek
from langchain_core.tools import tool
from langchain_core.messages import HumanMessage

# LangGraph 的核心组件
from langgraph.graph import MessagesState, StateGraph, START, END
from langgraph.prebuilt import ToolNode              # 自动执行工具的特殊节点
from langgraph.checkpoint.memory import MemorySaver  # 短期记忆存档器

# ==============================================================================
# 第二部分：初始化基础设施与定义工具
# ==============================================================================

# 1. 挂载短期记忆（记事本）
memory = MemorySaver()

# 2. 定义工具（AI 的“手脚”）
# 使用 @tool 装饰器，告诉 LangChain 这是一个可以让 AI 调用的外部函数
@tool
def search(query: str):
    """调用此函数可以浏览网络。""" 
    # 注意：函数的 Docstring（文档字符串）非常重要！大模型就是靠读这句话来知道这个工具是干嘛的。
    # 这里我们用硬编码模拟一个网络搜索结果
    return "北京天气晴朗 大约22度 湿度30%"

# 把定义好的工具装进列表里（以后可以加更多工具，比如计算器、发邮件等）
tools = [search]

# 3. 创建一个“工具执行机”节点（ToolNode）
# 这是一个预置节点，当 AI 决定用工具时，它会自动把参数提取出来，并运行上面的 search 函数。
tool_node = ToolNode(tools)

# 4. 初始化大模型大脑
model = ChatDeepSeek(
    model="Pro/deepseek-ai/DeepSeek-V3",
    temperature=0,
    api_key=os.environ.get("DEEPSEEK_API_KEY"),
    base_url=os.environ.get("DEEPSEEK_API_BASE"),
)

# 【关键】把工具“绑定”给大模型。
# 这样大模型在思考时，就知道自己手里有哪些牌可以打（有哪些工具可以调用）。
bound_model = model.bind_tools(tools)

# ==============================================================================
# 第三部分：定义节点函数与路由（大脑逻辑与交通警察）
# ==============================================================================

# 1. 定义交通警察（条件路由函数）
# 它的作用是：检查 AI 刚才的回答，决定流水线下一步往哪走。
def should_continue(state: MessagesState) -> Literal["action", END]:
    """返回下一个要执行的节点。"""
    # 拿到 AI 刚刚说的最后一条消息
    last_message = state["messages"][-1]
    
    # 如果没有 tool_calls（工具调用请求），说明 AI 觉得自己能回答，直接结束（走向 END）
    if not last_message.tool_calls:
        return END
        
    # 如果包含 tool_calls，说明 AI 需要查天气，让流水线走向 "action"（工具车间）
    return "action"

# 2. 定义大脑节点（Agent 节点）
def call_model(state: MessagesState):
    """调用大模型进行思考和回答"""
    # 把当前所有的聊天记录发给绑定了工具的模型
    response = bound_model.invoke(state["messages"])
    # 将模型的回复（可能是普通文本，也可能是工具调用请求）更新到状态列表中
    return {"messages": response}

# ==============================================================================
# 第四部分：搭建工作流图（拼装流水线）
# ==============================================================================

# 1. 拿出图纸，规定数据类型
workflow = StateGraph(MessagesState)

# 2. 添加两个核心节点
workflow.add_node("agent", call_model)  # 节点名："agent"（大脑，负责思考和决定）
workflow.add_node("action", tool_node)  # 节点名："action"（双手，负责执行工具）

# 3. 设置入口点：程序一启动，先进入 "agent" 节点思考
workflow.add_edge(START, "agent")

# 4. 添加条件边（核心逻辑）：从 "agent" 出来后，交给交通警察 should_continue 判断
workflow.add_conditional_edges(
    "agent",
    should_continue,
    # 路径映射：根据 should_continue 的返回值，决定去哪个节点
    ["action", END] # 如果返回 "action" 就去工具节点，返回 END 就结束
)

# 5. 添加回环边（魔法圈）：工具执行完后，必须把结果送回给 "agent" 大脑
# 让大脑看着工具的返回结果，组织语言再回答用户
workflow.add_edge("action", "agent")

# 6. 编译流水线，并装配上短期记忆（checkpointer）
# 注意：这里没有设置长期记忆（没有加 store）
app = workflow.compile(checkpointer=memory)

# ==============================================================================
# 第五部分：运行与测试
# ==============================================================================

# 1. 设置通行证（thread_id 必须一致，AI 才能记住上下文）
config = {"configurable": {"thread_id": "20"}}

# 2. 第一次对话：告诉 AI 名字
input_message = HumanMessage(content="hi! 我是tomie")
print("--- 第一次对话 ---")
# 使用 stream 进行流式输出
for event in app.stream({"messages": [input_message]}, config, stream_mode="values"):
    # 打印最新的一条消息
    event["messages"][-1].pretty_print()

# 3. 第二次对话：考验记忆
input_message = HumanMessage(content="我叫什么名字?")
print("\n--- 第二次对话 ---")
# 依然使用同一个 config
for event in app.stream({"messages": [input_message]}, config, stream_mode="values"):
    event["messages"][-1].pretty_print()
```

### 📝 总结

*   **Agent 与传统对话的区别**：
    普通对话是“直线”：`用户 -> 模型 -> 回答`。
    Agent 是“循环”：`用户 -> 模型思考（发现缺信息） -> 调用工具 -> 模型结合工具结果再思考 -> 回答`。
*   **三大核心魔法部件**：
    1.  `bind_tools(tools)`：给大脑发说明书，告诉它有哪些工具可用。
    2.  `add_conditional_edges`：设立红绿灯（路由），决定 AI 是直接闭嘴（`END`）还是去用工具（`action`）。
    3.  `add_edge("action", "agent")`：强制闭环，工具干完活必须把数据交还给大脑进行总结。
*   **`ToolNode` 的妙用**：它是个全自动执行器。AI 给出调用指令，`ToolNode` 自动解析指令、填入参数、运行代码、打包结果。你不用写任何解析 JSON 的恶心代码。
