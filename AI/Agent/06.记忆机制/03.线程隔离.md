# 线程隔离

在代码里，通常就是生成一个唯一的 `thread_id`，把存档严格隔离开。

```python
# ==========================================
# 第一步：准备工具箱（导入所需的库）
# ==========================================
import os
from langchain_deepseek import ChatDeepSeek
from langgraph.graph import StateGraph, MessagesState, START
# 核心：导入内存存档器（为了给AI植入短期记忆）
from langgraph.checkpoint.memory import MemorySaver 


# ==========================================
# 第二步：配置 AI 大脑（初始化大模型）
# ==========================================
model = ChatDeepSeek(
    model="Pro/deepseek-ai/DeepSeek-V3",        # 选择使用的模型版本
    temperature=0,                              # 温度设为0，让回答严谨、不瞎编（适合代码和逻辑）
    api_key=os.environ.get("DEEPSEEK_API_KEY"), # 从环境变量读取你的 API 密钥（注意保密！）
    base_url=os.environ.get("DEEPSEEK_API_BASE")# 接口请求的基础地址
)


# ==========================================
# 第三步：定义工位职责（编写节点函数）
# ==========================================
def call_model(state: MessagesState):
    # state 里面装的是当前的聊天记录（历史消息）
    # 把聊天记录打包发给 AI 模型去思考
    response = model.invoke(state["messages"]) 
    # 把 AI 的回答作为新消息，更新到聊天记录（状态）中返回
    return {"messages": response} 


# ==========================================
# 第四步：搭建流水线（构建并编译图）
# ==========================================
# 1. 拿出空白图纸，规定流水线上的货物是“聊天状态(Messages)”
builder = StateGraph(MessagesState)

# 2. 在图纸上画一个“工位”，名字叫 call_model，干活的就是上面写的函数
builder.add_node("call_model", call_model)

# 3. 画一条线，规定程序一启动（START），就直奔这个工位
builder.add_edge(START, "call_model")

# 4. 准备一个“内存存档器”（这就是我们要加的“短期记忆”核心组件）
memory = MemorySaver()

# 5. 组装！把图纸变成真正的机器，并且【重点】把存档器装上。
# 这里的 checkpointer 就是让 AI 拥有“记忆”的秘密武器
graph = builder.compile(checkpointer=memory) 


# ==========================================
# 第五步：开始对话测试（带上会话凭证）
# ==========================================
# 【重点】配置会话凭证（thread_id）。
# 它相当于给这次聊天开了一个专属“档案袋”。只要 thread_id 一样，AI 就知道是同一个对话
config = {"configurable": {"thread_id": "chat_room_1"}} 

# ---------- 第一次对话 ----------
# 准备第一句话
input_message = {"role": "user", "content": "hi! 我是tomie"} 

# 把消息扔进流水线。注意：要把 config 传进去，告诉它存到哪个档案袋
# stream_mode="values" 意思是每走一步，就把当前的所有状态输出出来（流式打印）
for chunk in graph.stream({"messages": [input_message]}, config, stream_mode="values"):
    # chunk["messages"] 是当前的所有历史消息列表
    # [-1] 取最后一条（也就是最新生成的那条 AI 消息）
    # pretty_print() 把它漂亮地打印在控制台上
    chunk["messages"][-1].pretty_print() 

# 运行到这里，AI 会回答：“你好 tomie...” 
# （此时，“tomie”这个名字已经被存入了 thread_id 为 chat_room_1 的档案袋里）


# ---------- 第二次对话 ----------
# 准备第二句话，故意考验它
input_message = {"role": "user", "content": "我叫什么名字?"} 

# 再次扔进流水线，【关键】依然使用同一个 config（同一个档案袋）
for chunk in graph.stream({"messages": [input_message]}, config, stream_mode="values"):
    chunk["messages"][-1].pretty_print()

# 运行到这里，因为有了存档器，AI 会读取档案，聪明地回答：“你叫 tomie 呀！”
```

### 📝 笔记必记的三个核心知识点（Cheat Sheet）：

1.  **`MemorySaver()`**：相当于给 AI 的大脑插了一块**内存条**。它把聊天记录临时存在运行内存里（只要不关程序，它就记得）。
2.  **`checkpointer=memory`**：在 `compile()` 阶段安装存档器。相当于给流水线配备了**自动存档员**。
3.  **`config = {"configurable": {"thread_id": "..."}}`**：这是**打开对应档案柜的钥匙**。同一个 `thread_id` 就是同一个聊天上下文。如果你把第二句的 `config` 换成一个新的 ID（比如 `chat_room_2`），AI 就会重新变回失忆状态，回答“我不知道你叫什么”。
