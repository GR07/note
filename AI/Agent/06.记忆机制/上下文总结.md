# 摘要记忆

把早期对话浓缩成一段文字

利用 LLM 的能力，将传入的对话历史列表进行总结，并返回一个包含摘要的系统提示。

```python
from openai import OpenAI

# 初始化 OpenAI 客户端
client = OpenAI(
    api_key="sk-fxxx", 
    base_url="https://api.deepseek.com"
)

message_history = []

system_prompt = """
我是系统提示词
"""

message_history.append({"role": "system", "content": system_prompt})

message_history.append({"role": "user", "content": "用户提示词1"})
message_history.append({"role": "user", "content": "用户提示词2"})
message_history.append({"role": "user", "content": "用户提示词3"})
message_history.append({"role": "user", "content": "用户提示词4"})
def context_summary(message_list):
    # 调用大模型
    response = client.chat.completions.create(
        model="deepseek-flash",
        messages=[
            {
                "role": "user", 
                # 构造一个提示词，要求模型对 message_list 中的内容进行总结
                "content": f"请对以下对话内容进行总结: {message_list}" 
            }
        ],
    )

    # 提取模型返回的总结内容，并封装成一个 system 角色的新消息列表
    new_message_list = [{"role": "system", "content": f"此前对话内容的总结: {response.choices[0].message.content}"}]
    
    # 返回这个新的、简短的消息列表
    return new_message_list

# 调用函数并打印结果
print(context_summary(message_history))
```
