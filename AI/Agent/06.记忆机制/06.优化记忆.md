# 两种方案

- 消息过滤，直接删除/截断旧消息

- 消息总结，让一个小模型去把前面 50 句对话浓缩成一句

**记忆优化要点：** 做到召回率与精度的平衡（多与准的权衡）




# 方案 - 消息过滤

加一个过滤函数处理 filter_messages

```python
import os
from typing import Literal
from langchain_deepseek import ChatDeepSeek
from langchain_core.tools import tool
from langgraph.checkpoint.memory import MemorySaver
from langgraph.graph import MessagesState, StateGraph, START, END
from langgraph.prebuilt import ToolNode

# 1. 记忆与工具准备
memory = MemorySaver()

@tool
def search(query: str):
    """调用此函数可以浏览网络。"""
    return "北京天气晴朗 大约22度 湿度30%"

tools = [search]
tool_node = ToolNode(tools)

# 2. 初始化模型并绑定工具
model = ChatDeepSeek(
    model="Pro/deepseek-ai/DeepSeek-V3",
    temperature=0,
    api_key=os.environ.get("DEEPSEEK_API_KEY"),
    base_url=os.environ.get("DEEPSEEK_API_BASE"),
)
bound_model = model.bind_tools(tools)

# 3. 定义交通警察（条件路由）
def should_continue(state: MessagesState):
    """返回下一个要执行的节点。"""
    last_message = state["messages"][-1]
    if not last_message.tool_calls:
        return END
    return "action"

# 4. 【核心优化】定义过滤函数
def filter_messages(messages: list):
    # 极其简单粗暴：只保留最后一条消息，防止上下文撑爆
    # 实际项目中，可以写更复杂的逻辑，比如只保留最近5条，或者保留带有特定标记的消息
    return messages[-1:]

# 5. 定义大脑节点
def call_model(state: MessagesState):
    # 使用过滤后的消息（最后一条）去请求大模型
    messages = filter_messages(state["messages"])
    response = bound_model.invoke(messages)
    return {"messages": response}

# 6. 搭建流水线
workflow = StateGraph(MessagesState)
workflow.add_node("agent", call_model)
workflow.add_node("action", tool_node)

workflow.add_edge(START, "agent")

# 条件边：大脑思考后，决定是用工具还是结束
workflow.add_conditional_edges(
    "agent",
    should_continue,
    ["action", END]
)

# 循环边：工具用完后，结果返回给大脑再思考一次
workflow.add_edge("action", "agent")

# 7. 编译（装上短期记忆）
app = workflow.compile(checkpointer=memory)

# 8. 测试运行
config = {"configurable": {"thread_id": "1"}}
# 注意：因为这里只传最后一条消息，所以哪怕有 memory，AI 也大概率不记得你第一句说了啥
# 这就是“过滤”带来的代价（牺牲了历史记忆，换取了上下文安全）
response = app.invoke(
    {"messages": [("human", "北京天气怎么样？")]}, 
    config
)
response["messages"][-1].pretty_print()
```


# 方案 - 消息总结

每次聊天快超标时，秘书就出来把前文写成一份“会议纪要（Summary）”，然后把冗长的原始记录用碎纸机（RemoveMessage）销毁，只保留这份极短的纪要和最新的两句话。


结合了“状态自定义”、“条件路由”、“消息删除”和“上下文压缩”。

```python
# ==============================================================================
# 第一部分：导入所需的工具箱
# ==============================================================================
import os
from typing import Literal, Annotated
from typing_extensions import TypedDict

from langchain_deepseek import ChatDeepSeek
# 导入消息类型：SystemMessage(系统提示), HumanMessage(用户消息), RemoveMessage(删除指令)
from langchain_core.messages import SystemMessage, HumanMessage, RemoveMessage

# LangGraph 核心组件
from langgraph.graph import StateGraph, START, END
# add_messages 是一个 Reducer（ reducer ），告诉框架如何更新 messages 列表（追加模式）
from langgraph.graph.message import add_messages
from langgraph.checkpoint.memory import MemorySaver


# ==============================================================================
# 第二部分：自定义状态 (State)
# ==============================================================================
# 大白话：以前我们直接用 MessagesState，现在我们需要额外加一个“会议纪要”字段
class State(TypedDict):
    # 这里的 Annotated[list, add_messages] 非常关键！
    # 它告诉 LangGraph：每次节点返回新的 messages 时，不要覆盖旧的，而是用 add_messages 函数把它们【追加】到列表后面。
    # 同理，如果返回的是 RemoveMessage，它会自动从列表中删除对应的消息。
    messages: Annotated[list, add_messages]
    # 新增字段：专门用来存放之前对话的“摘要”，默认为空字符串
    summary: str 


# ==============================================================================
# 第三部分：初始化大模型
# ==============================================================================
model = ChatDeepSeek(
    model="Pro/deepseek-ai/DeepSeek-V3",
    temperature=0,
    api_key=os.environ.get("DEEPSEEK_API_KEY"),
    base_url=os.environ.get("DEEPSEEK_API_BASE"),
)


# ==============================================================================
# 第四部分：定义节点函数 (核心逻辑)
# ==============================================================================

# 1. 正常聊天节点（AI 大脑）
def call_model(state: State):
    # 尝试从状态中获取之前留下的“会议纪要”，如果没有则返回空字符串
    summary = state.get("summary", "")
    
    # 如果存在纪要（说明之前长对话被压缩过）
    if summary:
        # 将纪要包装成系统指令（SystemMessage），提醒 AI 之前聊了什么
        system_message = f"之前对话的摘要：{summary}"
        # 将系统指令放在最前面，拼接上当前的全部聊天记录，准备发给大模型
        messages = [SystemMessage(content=system_message)] + state["messages"]
    else:
        # 如果没有纪要，说明是刚开场，直接发送聊天记录
        messages = state["messages"]
    
    # 调用大模型进行思考，传入整理好的 messages
    response = model.invoke(messages)
    
    # 返回一个字典，更新 state。
    # 注意：由于前面定义了 add_messages，这里返回的 response 会被自动【追加】到 messages 列表的末尾
    return {"messages": [response]}


# 2. 总结秘书节点（负责写纪要、删旧消息）
def summarize_conversation(state: State):
    # 再次获取现有的纪要（可能是空的）
    summary = state.get("summary", "")
    
    # 构建提示词：决定是“从头写”还是“扩展写”
    if summary:
        # 如果之前已经有纪要了，提示 AI 把新聊天内容合并进去，扩展摘要
        summary_message = (
            f"这是迄今为止对话的摘要：{summary}\n\n"
            "考虑上面的新消息，扩展摘要："
        )
    else:
        # 如果之前没纪要，提示 AI 直接总结刚才的对话
        summary_message = "创建上述对话的摘要："
    
    # 把写好的提示词打包成 HumanMessage，放到现有的聊天记录最后面，一起发给大模型
    # （大模型需要看到提示词才知道自己要干嘛）
    messages = state["messages"] + [HumanMessage(content=summary_message)]
    
    # 大模型开工，生成摘要内容
    response = model.invoke(messages)
    
    # 【核心魔法：碎纸机启动】
    # state["messages"][:-2] 意思是：获取列表里除了最后两条之外的所有消息。
    # 列表推导式：为每一条要删除的消息，创建一个 RemoveMessage 对象，并带上它原本的 ID。
    delete_messages = [RemoveMessage(id=m.id) for m in state["messages"][:-2]]
    
    # 返回更新：把新生成的摘要字符串存进 summary 字段；把删除指令发给 messages 列表（触发删除）
    return {"summary": response.content, "messages": delete_messages}


# 3. 交通警察节点（条件路由）
def should_continue(state: State) -> Literal["summarize_conversation", END]:
    # 获取当前的完整聊天记录列表
    messages = state["messages"]
    
    # 设定阈值：如果消息数量超过了 6 条（意味着积累了太多的旧消息）
    if len(messages) > 6:
        # 返回字符串 "summarize_conversation"，告诉流水线去总结节点
        return "summarize_conversation"
    
    # 如果没有超过 6 条，无需总结，直接结束当前这一轮对话
    return END


# ==============================================================================
# 第五部分：搭建并编译流水线 (建厂)
# ==============================================================================

# 1. 初始化状态图，告诉它我们的数据结构是刚刚自定义的 State
workflow = StateGraph(State)

# 2. 添加节点（把函数装进工位）
workflow.add_node("conversation", call_model)                     # 工位 A：聊天
workflow.add_node("summarize_conversation", summarize_conversation) # 工位 B：总结与清理

# 3. 设置入口点：程序启动先去找 AI 聊天
workflow.add_edge(START, "conversation")

# 4. 添加条件边：从聊天节点出来后，由 should_continue 决定去哪
workflow.add_conditional_edges(
    "conversation",       # 起始节点
    should_continue,      # 决策函数（交警）
    # 路径映射：告诉框架，决策函数可能返回的两个结果分别对应哪些节点
    ["summarize_conversation", END] 
)

# 5. 添加普通边：总结节点干完活后，直接结束这轮流程
workflow.add_edge("summarize_conversation", END)

# 6. 编译流水线，并挂载短期记忆存档器
memory = MemorySaver()
app = workflow.compile(checkpointer=memory)


# ==============================================================================
# 第六部分：测试运行
# ==============================================================================
if __name__ == "__main__":
    # 设置会话 ID，确保使用同一份记忆
    config = {"configurable": {"thread_id": "test_summary_1"}}
    
    # 模拟连续对话，触发总结机制
    # 你可以循环发 7-8 次消息，当消息数量超过 6 时，会自动触发总结节点，控制上下文长度。
    for i in range(7):
        user_input = input(f"第{i+1}轮 - 请输入内容: ")
        if user_input.lower() in ['quit', 'exit']:
            break
            
        print("\n--- AI 正在思考... ---")
        # 使用 stream 流式打印过程
        for event in app.stream({"messages": [("human", user_input)]}, config, stream_mode="values"):
            # 打印最后一条消息
            event["messages"][-1].pretty_print()
        
        # 如果你想知道当前的状态里存了什么（摘要和列表长度），可以打印出来
        state = app.get_state(config).values
        print(f"\n[调试信息] 当前 summary: {state.get('summary', '无')}")
        print(f"[调试信息] 当前列表中的消息数: {len(state['messages'])}")
        print("-" * 50)
```

### 📝 笔记重点标注

1.  **`Annotated[list, add_messages]`**：这是 LangGraph 里处理列表的“神级操作”。它让你的节点函数只需 `return {"messages": [新消息]}`，框架会自动把新消息**追加**进去，而不是覆盖整个列表。如果返回的是 `RemoveMessage`，它还会自动执行删除。
2.  **`RemoveMessage`**：它是**唯一**能从内存（State）中真正移除历史记录的方法。这对于节省 Token 至关重要。
3.  **切片 `[:-2]`**：切掉最后两个元素，保留最新的两条，这是一种常见的保留最新上下文（缓冲窗口）的策略。
4.  **摘要的传递**：注意 `call_model` 里把 `summary` 塞进 `SystemMessage` 的操作，这就是把长期记忆塞给大模型大脑的最标准姿势。
