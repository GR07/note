

# 主流程

- 定义多个Agent，各司其职

- 定义各种工具 Tool

- 定义任务 Task

- 组建多智能体协作团队（Crew）

- 注入业务数据并启动执行


```python

"""
CrewAI B2B 销售线索自动化挖掘系统
=================================
基于多智能体协作，自动完成客户背调、需求分析与个性化触达策略生成。
"""

import os
from dotenv import load_dotenv

# ── 1. 加载环境变量 ──────────────────────────────────────────
load_dotenv()  # 从 .env 文件加载 API 密钥

# ── 2. 导入 CrewAI 核心组件 ──────────────────────────────────
from crewai import Agent, Task, Crew, LLM          # 核心框架
from crewai_tools import (                          # 工具集
    DirectoryReadTool,   # 读取目录内容
    FileReadTool,        # 读取单个文件
    SerperDevTool,       # 联网搜索（基于 Serper API）
    WebsiteSearchTool,   # 网页内容检索（可选，用于补充信息）
)

# ── 3. 配置 LLM（可选，默认使用 gpt-4o-mini）───────────────
# 可通过环境变量 OPENAI_MODEL_NAME 指定模型，
# 也可直接通过 LLM 类设置 temperature、base_url 等参数[reference:2][reference:3]
llm = LLM(
    model=os.getenv("OPENAI_MODEL_NAME", "gpt-4o-mini"),
    temperature=0.2,          # 较低温度 → 输出更聚焦、少发散
    # base_url="https://api.openai.com/v1",  # 如需代理/兼容端点可解除注释
)

# ── 4. 实例化工具 ────────────────────────────────────────────
# DirectoryReadTool 指定读取的项目说明目录，
# FileReadTool 不传参时默认读取任意指定文件，
# SerperDevTool 依赖环境变量 SERPER_API_KEY[reference:4]
directory_read_tool = DirectoryReadTool(directory="./instructions")
file_read_tool      = FileReadTool()
search_tool         = SerperDevTool()
# web_search_tool     = WebsiteSearchTool()  # 可选：抓取网页正文

# ── 5. 定义 Agent（智能体）───────────────────────────────────
# 5.1 销售代表：负责执行一线的客户背调与信息搜集
sales_rep_agent = Agent(
    role="销售代表",
    goal="识别符合我们理想客户画像的高价值潜在客户",
    backstory=(
        "作为AI大模型智能客服产品销售团队的一员，"
        "你的任务是搜索数字化环境中的潜在客户。"
        "你拥有尖端工具和战略思维，分析数据、趋势和互动，"
        "发掘其他人可能忽略的机会。"
        "你的工作对推动有意义的互动以及公司增长至关重要。"
    ),
    allow_delegation=False,   # 禁止将任务委派给其他 Agent，确保独立执行
    verbose=True,
    llm=llm,                  # 绑定 LLM 配置
    tools=[directory_read_tool, file_read_tool, search_tool],  # 为该 Agent 挂载工具
)

# 5.2 销售主管：负责基于背调结果生成个性化触达策略
lead_sales_rep_agent = Agent(
    role="销售主管",
    goal="基于销售代表提供的客户背调报告，制定个性化触达策略并生成触达内容",
    backstory=(
        "你是AI大模型智能客服产品销售团队的主管，"
        "擅长将深度洞察转化为有影响力的客户沟通方案。"
        "你审阅销售代表提交的客户资料，提炼关键信息，"
        "为每一位潜在客户量身定制开场白、价值主张与互动节奏。"
    ),
    allow_delegation=False,
    verbose=True,
    llm=llm,
    tools=[directory_read_tool, file_read_tool],  # 主管主要依赖阅读能力
)

# ── 6. 定义 Task（任务）─────────────────────────────────────
# 6.1 客户背调任务：由 sales_rep_agent 执行
lead_profiling_task = Task(
    description=(
        "使用中文输出。"
        "对{lead_name}进行深入分析。"
        "该公司属于{industry}行业，最近对我们的解决方案表现出兴趣。"
        "利用所有可用的数据来源，编写详细的客户资料。"
        "重点关注关键决策者、近期业务发展，以及与我们产品相匹配的潜在需求。"
        "此任务对有效制定我们的互动策略至关重要。\n"
        "不要做任何假设，只使用你绝对确信的信息。"
    ),
    expected_output=(
        "使用中文输出。"
        "关于{lead_name}的综合报告："
        "包括公司背景、关键人员、近期里程碑和已识别需求。"
        "突出我们解决方案可以提供价值的潜在领域，"
        "并提出个性化的互动策略建议。"
    ),
    agent=sales_rep_agent,
    tools=[directory_read_tool, file_read_tool, search_tool],
)

# 6.2 个性化触达任务：由 lead_sales_rep_agent 执行
#     依赖 lead_profiling_task 的输出作为上下文
personalized_outreach_task = Task(
    description=(
        "使用中文输出。"
        "基于销售代表提交的{lead_name}背调报告，"
        "为关键决策者{key_decision_maker}（职位：{position}）"
        "撰写一封个性化的触达邮件或消息。"
        "内容需包含："
        "1）以{lead_name}近期里程碑（{milestone}）为切入点的开场；"
        "2）清晰阐述我们的解决方案如何匹配其潜在需求；"
        "3）一个明确的下一步行动邀约（如15分钟电话会议）。"
        "语气专业、真诚，避免过度推销。"
    ),
    expected_output=(
        "使用中文输出。"
        "一封完整的个性化触达邮件/消息草稿，"
        "包含主题、正文与行动号召，"
        "并附上 3 条针对该客户的沟通要点摘要。"
    ),
    agent=lead_sales_rep_agent,
    tools=[directory_read_tool, file_read_tool],
    # 关键：让本任务读取上一任务的输出作为上下文
    context=[lead_profiling_task],
)

# ── 7. 组装 Crew（协作团队）─────────────────────────────────
# memory=True 开启团队记忆，使多个 Agent 在处理同一批客户时信息互通[reference:5]
crew = Crew(
    agents=[sales_rep_agent, lead_sales_rep_agent],
    tasks=[lead_profiling_task, personalized_outreach_task],
    verbose=2,       # 详细日志级别（0=静默，1=标准，2=详细）
    memory=True,     # 开启短期与长期记忆，避免重复检索
    llm=llm,         # 团队级默认 LLM
)

# ── 8. 注入业务数据并启动执行 ────────────────────────────────
inputs = {
    "lead_name": "MoPaaS 魔泊云",
    "industry": "人工智能科技公司",
    "key_decision_maker": "鲁为民",
    "position": "CEO",
    "milestone": "C端大模型课程发布",
}

if __name__ == "__main__":
    print("🚀 正在启动销售线索自动化调研...\n")
    result = crew.kickoff(inputs=inputs)   # 同步执行，返回 CrewOutput 对象[reference:6]
    print("\n" + "=" * 60)
    print("✅ 执行完成，最终输出：\n")
    print(result)
```
