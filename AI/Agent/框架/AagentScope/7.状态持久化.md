# 状态持久化（记忆保存）

以前的 Agent 一旦关闭终端，所有的对话记忆就丢了。现在通过 agent.state.model_dump(mode="json")，可以把当前的对话历史、Token 计数、系统状态等全部存到 agent_state.json 里。下次运行代码时，读取这个文件并传给 Agent(state=...)，AI 就能无缝接续之前的对话。


```python
# ==========================================
# 1. 导入依赖库
# ==========================================
import asyncio  # 导入异步 I/O 库，用于支持异步编程
from agentscope.agent import Agent, ContextConfig  # 导入 Agent 核心类和上下文配置类
from agentscope.workspace import LocalWorkspace  # 导入本地工作区类 # 导入本地工作区
from agentscope.model import OpenAIChatModel  # 导入大语言模型类
from agentscope.credential import OpenAICredential  # 导入凭证配置类
from agentscope.message import Msg, TextBlock  # 导入消息类和文本块类
from agentscope.tool import Toolkit  # 导入工具集类
from agentscope.state import AgentState  # 导入 Agent 状态类（核心：用于保存/加载记忆）
from agentscope.event import (RequireUserConfirmEvent, ConfirmResult,  # 导入需要用户确认的事件类、确认结果类
                              UserConfirmResultEvent)  # 导入用户确认结果事件类
import json  # 导入 JSON 库，用于序列化/反序列化状态数据
import os  # 导入 OS 库，用于判断本地文件是否存在

# ==========================================
# 2. 核心主函数
# ==========================================
async def basic_usage():  # 定义异步主函数
    # --- 初始化工作区 ---
    workspace = LocalWorkspace(workdir="./my_agent_workspace")  # 实例化本地工作空间，指定运行目录
    await workspace.initialize()  # 异步初始化工作区
    tools = await workspace.list_tools()  # 获取工作区默认提供的工具列表
    
    # --- 配置上下文管理 ---
    contextconfig = ContextConfig(  # 实例化上下文配置对象
        trigger_ratio=0.8,  # 触发压缩阈值：达到 80% Token 时触发
        reserve_ratio=0.1,  # 保留比例：压缩时保留最近 10% 的对话
        tool_result_limit=3000  # 工具结果限制：单次工具调用返回不超过 3000 字符
    )
    
    # --- 初始化模型 ---
    model = OpenAIChatModel(  # 实例化大模型
        credential=OpenAICredential(  # 配置凭证
            api_key="xxx",  
            base_url="xxx", 
        ),
        model="qwen3.6-flash",  
    )

    # ==========================================
    # 3. 状态加载与 Agent 实例化（实现记忆持久化）
    # ==========================================
    if os.path.isfile("agent_state.json"):  # 如果本地存在状态文件（说明之前运行过并保存了记忆）
        with open("agent_state.json", "r") as f:  # 以只读模式打开文件
            loaded_state_dict = json.load(f)  # 将 JSON 数据加载为 Python 字典
        loaded_state = AgentState.model_validate(loaded_state_dict)  # 将字典解析并验证为 AgentState 对象
        
        agent = Agent(  # 实例化 Agent
            name="my_assistant",
            system_prompt="你是一个有帮助的助手。",
            model=model,
            toolkit=Toolkit(tools=tools),
            offloader=workspace,
            context_config=contextconfig,
            state=loaded_state,  # ⚠️ 关键：将加载的历史状态注入 Agent，恢复记忆
        )
    else:  # 如果本地不存在状态文件（首次运行）
        agent = Agent(  # 正常实例化 Agent（使用默认的空状态）
            name="my_assistant",
            system_prompt="你是一个有帮助的助手。",
            model=model,
            toolkit=Toolkit(tools=tools),
            offloader=workspace,
            context_config=contextconfig,
            # 这里没有传 state，Agent 内部会自动生成一个全新的 AgentState
        )

    # ==========================================
    # 4. 手动压缩上下文
    # ==========================================
    await agent.compress_context(  # 手动调用上下文压缩方法
        context_config=ContextConfig(  # 覆盖全局配置，使用更激进的压缩策略
            trigger_ratio=0.5,  # 达到 50% 就触发压缩
            reserve_ratio=0.1,  # 保留最近 10%
        )
    )  # 这通常在加载了非常大的历史状态文件后，或者开启新话题前调用

    # ==========================================
    # 5. 对话主循环
    # ==========================================
    confirm_event = None  # 初始化用户确认事件变量
    while True:  # 无限循环进行对话
        if confirm_event is None:  # 如果没有待确认的事件
            user_msg = input("user>>>")  # 等待用户输入
            user_msg = Msg(  # 将用户输入封装成标准的消息对象
                role="user",
                content=[TextBlock(type="text", text=user_msg)],
                name="user"
            )
        else:  # 如果有待确认的事件（比如工具执行需要授权）
            user_msg = confirm_event  # 将确认结果作为消息发给模型
            confirm_event = None  # 清空状态
            
        print("agent>>>")  # 打印提示符
        
        # --- 处理流式回复 ---
        async for event in agent.reply_stream(user_msg):  # 异步流式获取 Agent 响应
            # 分支 1：处理需要用户确认的工具调用
            if isinstance(event, RequireUserConfirmEvent):
                confirm_results = []
                for tc in event.tool_calls:  # 遍历需要确认的工具
                    # 弹出终端询问，根据用户输入 Y/y 判断是否同意
                    is_confirmed = input(f"\nsystem>>>是否同意调用{tc.name}(Y/N), {tc.input}:") in ["Y", "y"]
                    confirm_results.append(ConfirmResult(
                        confirmed=is_confirmed,
                        tool_call=tc,
                        rules=tc.suggested_rules
                    ))
                confirm_event = UserConfirmResultEvent(  # 构造用户确认结果事件
                    reply_id=event.reply_id,
                    confirm_results=confirm_results,
                )
                
            # 分支 2：处理正常的文本流式输出
            elif hasattr(event, "delta"):  # 如果事件带有文本增量
                print(event.delta, end="", flush=True)  # 打印文本片段（打字机效果）
        
        # ==========================================
        # 6. 状态保存（持久化记忆）
        # ==========================================
        # 在每轮对话结束后，将当前的完整状态序列化
        saved_state_dict = agent.state.model_dump(mode="json")  # 将 AgentState 转为 JSON 兼容的字典格式
        with open("agent_state.json", "w") as f:  # 以写入模式打开文件
            json.dump(saved_state_dict, f)  # 将字典保存到本地 JSON 文件中（覆盖旧文件）
            
        print("\n")  # 一轮对话结束，换行

# ==========================================
# 7. 程序入口
# ==========================================
if __name__ == "__main__":  # 确保脚本被直接运行时才执行
    asyncio.run(basic_usage())  # 启动异步事件循环，运行主函数

```
