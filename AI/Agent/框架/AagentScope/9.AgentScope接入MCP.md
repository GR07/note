
# 代码的核心功能与执行流程

关键流程如下：

1.  首先配置并启动了一个本地的 MCP 服务器（`mcp_server_stdio.py`），并通过 `MCPClient` 与它建立连接。这样，智能体就能“看到”并调用这个服务器上暴露的工具了。

2.  创建了一个 `Agent` 实例。

3.  启动人机协同的对话循环

4.  拦截并确认敏感操作


使用：对话询问有哪些mcp工具，然后指示做事情，模型会自动调用mcp得出结果返回。

```python
# ==========================================
# 1. 导入依赖库
# ==========================================
import asyncio  # 导入异步IO库，用于支持异步编程
from agentscope.agent import Agent  # 导入Agent类，用于创建智能体
from agentscope.model import OpenAIChatModel  # 导入OpenAI兼容的模型类
from agentscope.tool import Toolkit  # 导入工具包类，用于管理工具
from agentscope.mcp import StdioMCPConfig, HttpMCPConfig, MCPClient  # 导入MCP（模型上下文协议）相关配置和客户端
from agentscope.workspace import LocalWorkspace  # 导入本地工作区类
from agentscope.credential import OpenAICredential  # 导入凭证类，用于配置API Key
from agentscope.message import TextBlock, Msg  # 导入消息块和消息类，用于构建对话消息
from agentscope.event import RequireUserConfirmEvent, UserConfirmResultEvent, ConfirmResult  # 导入事件类，用于处理需要用户确认的交互事件

# ==========================================
# 2. 定义主异步函数及MCP配置
# ==========================================
async def main():
    # 配置通过标准输入输出（stdio）连接的MCP服务
    stdio_config = StdioMCPConfig(
        command="python",  # 运行命令：python
        args=["mcp_server_stdio.py"],  # 运行参数：目标Python脚本文件
    )
    
    # 注释掉的HTTP配置（通常用于连接远程MCP服务）
    # http_config = HttpMCPConfig(
    #     url="http://localhost:8000/mcp",
    # )
    
    # 初始化MCP客户端列表
    mcp_clients = [
        MCPClient(  # 实例化本地MCP客户端
            mcp_config=stdio_config,  # 传入上述stdio配置
            name="local_mcp",  # 客户端名称
            is_stateful=True),  # 设置为有状态（保持会话连接）
        
        # 注释掉的远程MCP客户端配置
        # MCPClient(
        #     mcp_config=http_config,
        #     name="remote_mcp",
        #     is_stateful=True),
    ]

    # ==========================================
    # 3. 初始化工作区与工具包
    # ==========================================
    # 创建本地工作区实例
    workspace = LocalWorkspace(
        workdir="./workspace",  # 指定工作目录
        default_mcps=mcp_clients,  # 将MCP客户端列表注入工作区
        # skill_paths=["./your_skill_folder"] # 注释掉的技能路径配置
    )
    
    # 异步初始化工作区（加载MCP服务和配置）
    await workspace.initialize()
    
    # 创建工具包实例
    toolkit = Toolkit(
        tools=await workspace.list_tools(),  # 获取工作区中所有可用的工具
        mcps=await workspace.list_mcps(),  # 获取工作区中所有可用的MCP
        skills_or_loaders=await workspace.list_skills(),  # 获取工作区中所有可用的技能
    )

    # ==========================================
    # 4. 配置大模型与创建Agent
    # ==========================================
    # 配置大模型的API凭证
    credential = OpenAICredential(
        api_key="xxx",  
        base_url="xxx",  
    )
    
    # 实例化大语言模型
    model = OpenAIChatModel(
        model="qwen3.7-plus",  # 使用的模型名称
        credential=credential,  # 传入凭证
        stream=True,  # 开启流式输出（打字机效果）
    )
    
    # 创建Agent实例
    agent = Agent(
        name="My_Agent",  # Agent的名称
        model=model,  # 传入大模型实例
        system_prompt="你是一个中文AI，回复过程中请输出标准的中文。",  # 设置系统提示词，规范AI输出语言
        toolkit=toolkit,  # 传入工具包，使Agent具备使用工具的能力
    )

    # ==========================================
    # 5. 主交互循环与事件处理
    # ==========================================
    confirm_event = None  # 用于暂存用户确认事件的变量
    
    while True:  # 开启无限循环，保持对话
        # 如果没有待确认的事件，则获取用户输入
        if confirm_event is None:
            user_msg = input("user>>> ")  # 从控制台读取用户输入
            # 将用户输入包装成标准的Msg对象
            user_msg = Msg(
                role="user",  # 角色为user
                content=[TextBlock(type="text", text=user_msg)],  # 内容为文本块
                name="user"  # 发送者名称
            )
        else:
            # 如果有待确认的事件，直接将确认事件作为下一条消息传给Agent
            user_msg = confirm_event
            confirm_event = None  # 清空待确认事件

        print("agent>>> ")  # 打印Agent的回复前缀
        
        # 异步遍历Agent的流式回复事件
        async for event in agent.reply_stream(user_msg):
            # 如果事件是需要用户确认工具调用
            if isinstance(event, RequireUserConfirmEvent):
                confirm_results = []  # 初始化确认结果列表
                for tc in event.tool_calls:  # 遍历需要确认的每一个工具调用
                    # 在控制台询问用户是否允许执行该工具，并封装结果
                    confirm_results.append(ConfirmResult(
                        confirm=(
                            # 提示用户是否同意执行
                            input(f"***system*** 是否同意调用 {tc.name}({tc.input})？\n 输入 ['y','Y']: ")
                            in ["y", "Y"]  # 判断输入是否为 y 或 Y
                        ),
                        rules=tc.suggested_rules,  # 传入建议的规则
                    ))
                # 构建用户确认结果事件
                confirm_event = UserConfirmResultEvent(
                    reply_id=event.reply_id,  # 关联原事件的ID
                    confirm_results=confirm_results,  # 传入上述确认结果
                )
            # 如果事件是流式输出的文本增量（delta）
            elif hasattr(event, "delta"):
                # 实时打印增量文本，不换行并强制刷新缓冲区
                print(event.delta, end="", flush=True)
        
        print("\n")  # 每轮回复结束后打印一个换行符

# ==========================================
# 6. 程序入口
# ==========================================
if __name__ == "__main__":
    asyncio.run(main())  # 运行主异步函数

```
