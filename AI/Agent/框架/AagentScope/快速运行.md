# AgentScope

AgentScope 是阿里巴巴通义实验室开源的一款生产级智能体（Agent）开发框架

早期它主要是一个以开发者为核心的多智能体开发框架。但从 2.0 版本开始，它演进为一套通用的 Agent Harness（工程化底座），核心面向企业级分布式智能体场景。


```python
# ==========================================
# 1. 导入依赖库
# ==========================================
from agentscope.message import Msg, TextBlock  # 导入消息处理模块：Msg（消息类）和 TextBlock（文本块类）
from agentscope.model import OpenAIChatModel  # 导入模型模块：用于创建兼容 OpenAI 接口的模型对象
from agentscope.agent import Agent  # 导入 Agent 模块：用于创建智能体（Agent）实例
import asyncio  # 导入 Python 内置的异步 I/O 库，用于支持异步编程
from agentscope.credential import OpenAICredential  # 导入凭证模块：用于管理 API 密钥和接口地址

# ==========================================
# 2. 配置凭证、模型与 Agent 实例化
# ==========================================
credential = OpenAICredential(  # 实例化一个 OpenAI 凭证对象
    api_key="xxx",  # 设置 API Key
    base_url="xxx"  # 设置 API 请求地址
)  # 结束凭证对象的配置

model = OpenAIChatModel(  # 实例化一个聊天模型对象
    model="qwen3.6-plus",  # 指定要调用的具体模型名称，这里使用的是通义千问 qwen3.6-plus 模型
    credential=credential,  # 传入前面定义好的凭证对象（包含 API Key 和 URL）
    stream=True  # 开启流式输出（True 表示模型会边生成边返回结果，而不是等全部生成完才一次性返回）
)  # 结束模型对象的配置

agent = Agent(  # 实例化一个智能体（Agent）对象
    name="My_Agent",  # 为这个智能体命名，名为 "My_Agent"
    model=model,  # 绑定前面配置好的大语言模型
    system_prompt="你是一个聪明的中文AI。"  # 设置系统提示词（System Prompt），定义 AI 的角色和行为设定
)  # 结束智能体对象的配置

# ==========================================
# 3. 流式对话逻辑（异步函数）
# ==========================================
async def chat_with_agent_stream():  # 定义一个异步函数，用于处理流式对话逻辑
    while True:  # 开启一个无限循环，使得对话可以持续进行，直到用户手动结束
        user_msg = input("user>>>")  # 在控制台打印 "user>>>" 并等待用户输入文本，将输入内容赋值给 user_msg 变量
        
        user_msg = Msg(  # 将用户的输入封装成一个标准的消息对象（Msg）
            role="user",  # 设定该消息的角色为 "user"（用户）
            content=[TextBlock(type="text", text=user_msg)],  # 设定消息内容，使用 TextBlock 封装，类型为 "text"，文本为用户刚刚输入的内容
            name="user"  # 设定发送该消息的实体名称为 "user"
        )  # 结束用户消息的封装
        
        print("agent>>>")  # 在控制台打印 "agent>>>"，提示接下来将输出 AI 的回复
        
        async for event in agent.reply_stream(user_msg):  # 使用异步迭代器接收流式返回的每一个事件（片段）
            if hasattr(event, "delta"):  # 检查当前事件对象中是否包含 "delta" 属性（delta 代表当前新增的文本片段）
                # 图片中写的是 "event delta" 存在拼写错误，这里修正为 "event.delta"
                print(event.delta, end="", flush=True)  # 打印当前文本片段，end="" 防止自动换行，flush=True 强制立即输出到控制台（实现打字机效果）
        
        print("\n")  # 当前一轮流式回复接收完毕后，打印一个换行符，以便与下一轮对话隔开

# ==========================================
# 4. 程序入口
# ==========================================
if __name__ == '__main__':  # Python 程序的执行入口判断，确保只有在直接运行该脚本时，下面的代码才会被执行
    asyncio.run(chat_with_agent_stream())  # 使用 asyncio 运行主异步函数 chat_with_agent_stream()，启动整个流式对话流程
```
