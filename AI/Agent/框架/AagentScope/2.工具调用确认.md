# 工具调用

工具调用前让用户确认的人机协同流程


用户消息 -> agent.reply_stream(...) -> RequireUserConfirmEvent -> 构造 UserConfirmResultEvent -> 作为下一轮输入回传 -> 继续流式输出 delta



```python
# ==========================================
# 1. 导入依赖库
# ==========================================
from agentscope.tool import Toolkit, FunctionTool, Read, ToolChunk  # 导入工具包相关类：Toolkit(工具管理器), FunctionTool(将普通函数转为工具), Read(内置读取工具), ToolChunk(工具返回结果块)
from agentscope.message import Msg, TextBlock  # 导入消息处理模块：Msg(消息类) 和 TextBlock(文本块类)
from agentscope.model import OpenAIChatModel  # 导入模型模块：用于创建兼容 OpenAI 接口的模型对象
from agentscope.agent import Agent  # 导入 Agent 模块：用于创建智能体（Agent）实例
import asyncio  # 导入 Python 内置的异步 I/O 库，用于支持异步编程
from agentscope.event import RequireUserConfirmEvent, ConfirmResult, UserConfirmResultEvent  # 导入事件模块：用于处理需要用户确认的工具调用事件及结果
from agentscope.credential import OpenAICredential  # 导入凭证模块：用于管理 API 密钥和接口地址

# ==========================================
# 2. 自定义工具 (Tool) 定义
# ==========================================
def add(a, b):  # 定义一个普通的 Python 函数，用来计算两个数字之和
    """  # 函数的文档字符串，大模型会读取这段描述来理解工具的作用
    计算两个数字之和的工具
    :param a: 第一个输入值
    :param b: 第二个输入值
    :return: 返回两个数相加的值
    """
    return ToolChunk(  # 将计算结果封装成 ToolChunk 对象返回给 Agent
        content=[TextBlock(type="text", text=str(int(a) + int(b)))]  # 将 a 和 b 转换为整型相加后，再转回字符串，并包装成文本块
    )  # 结束 ToolChunk 封装

# ==========================================
# 3. 配置工具集、凭证、模型与 Agent 实例化
# ==========================================
toolkit = Toolkit(tools=[Read(), FunctionTool(add)])  # 实例化工具集：包含一个内置的 Read（读取文件）工具，以及我们刚刚用 FunctionTool 包装的 add 函数

credential = OpenAICredential(  # 实例化一个 OpenAI 凭证对象
    api_key="xxx",  
    base_url="xxx"  
)  # 结束凭证对象的配置

model = OpenAIChatModel(  # 实例化一个聊天模型对象
    model="xxx",  # 指定要调用的具体模型名称
    credential=credential,  # 传入前面定义好的凭证对象
    stream=True  # 开启流式输出，模型会边生成边返回结果
)  # 结束模型对象的配置

agent = Agent(  # 实例化一个智能体（Agent）对象
    name="My_Agent",  # 为这个智能体命名
    model=model,  # 绑定前面配置好的大语言模型
    system_prompt="你是一个聪明的中文AI。",  # 设置系统提示词，定义 AI 的角色
    toolkit=toolkit  # 将前面配置好包含 Read 和 add 的工具集绑定给 Agent
)  # 结束智能体对象的配置

# ==========================================
# 4. 核心对话逻辑（包含工具确认机制）
# ==========================================
async def chat_with_agent_stream():  # 定义一个异步函数，处理流式对话和工具确认
    confirm_event = None  # 初始化确认事件变量为 None，用于暂存需要用户确认的事件
    
    while True:  # 开启无限循环，持续对话
        # 检查是否有需要确认的事件等待处理
        if confirm_event is None:  # 如果没有待确认事件
            user_msg = input("user>>>")  # 获取用户输入
            
            user_msg = Msg(  # 将用户输入封装成 Msg 对象
                role="user",  # 角色为用户
                content=[TextBlock(type="text", text=user_msg)],  # 消息内容为文本块
                name="user"  # 发送者名称
            )  # 结束用户消息封装
        else:  # 如果有待确认事件（说明上一轮 AI 请求调用工具，正在等用户批准）
            user_msg = confirm_event  # 将待确认事件直接作为本条发给模型的消息
            confirm_event = None  # 清空待确认状态，准备处理下一轮
            
        print("agent>>>")  # 打印提示符，准备输出 Agent 的回复
        
        # 异步接收 Agent 的流式返回事件
        async for event in agent.reply_stream(user_msg):  # 遍历模型返回的每一个事件
            # 分支 1：如果当前事件是要求用户确认工具调用
            if isinstance(event, RequireUserConfirmEvent):  # 判断事件类型
                confirm_results = []  # 初始化确认结果列表
                for tc in event.tool_calls:  # 遍历模型请求调用的所有工具 (tc = tool_call)
                    print(f"工具：{tc.name}, 输入：{tc.input}")  # 打印即将调用的工具名称和参数
                    print(f"建议规则：{tc.suggested_rules}")  # 打印建议的自动批准规则
                    
                    # 将用户的确认结果封装起来
                    confirm_results.append(ConfirmResult(  # 添加到结果列表
                        confirmed=True,  # True 表示同意调用，这里写死为 True；实际业务中这里可以改成 input() 让用户输入 Y/N (False 表示拒绝)
                        tool_call=tc,  # 将原始的 tool_call 对象传回
                        rules=tc.suggested_rules,  # 接受建议的规则以便未来自动允许相同操作
                    ))  # 结束 ConfirmResult 封装
                
                # 构造用户确认结果事件
                confirm_event = UserConfirmResultEvent(  
                    reply_id=event.reply_id,  # 传入对应的回复ID
                    confirm_results=confirm_results,  # 传入前面构造的确认结果列表
                )  # 结束 UserConfirmResultEvent 构造
                
            # 分支 2：如果当前事件包含文本增量（模型正在输出文字）
            elif hasattr(event, "delta"):  # 检查事件是否有 delta 属性
                print(event.delta, end="", flush=True)  # 打印文本片段，end="" 防止换行，flush=True 实现打字机效果
                
        print("\n")  # 一次流式回复结束后，打印换行符分隔

# ==========================================
# 5. 程序入口
# ==========================================
if __name__ == '__main__':  # 判断是否作为主程序运行
    asyncio.run(chat_with_agent_stream())  # 运行异步主函数，启动对话流
```
