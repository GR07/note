# 初始化配置文件 local_settings

```python
# 导入操作系统接口模块，主要用于读取环境变量
import os  
# 从 openai 库中导入异步客户端类 AsyncOpenAI
from openai import AsyncOpenAI  
# 从 agents 库的 models 模块中导入 openai_provider，用于后续配置模型
from agents.models import openai_provider  
# 从 agents 库中导入以下三个函数，用于全局配置
from agents import (  
    # 用于设置默认的 OpenAI API 接口类型
    set_default_openai_api,  
    # 用于设置默认的 OpenAI 客户端实例
    set_default_openai_client,  
    # 用于控制是否禁用追踪服务
    set_tracing_disabled,  
)

# 创建一个异步的 OpenAI 客户端实例，赋值给变量 client
client = AsyncOpenAI(  
    api_key="sk-fxxx",
    base_url="https://api.deepseek.com",
)

# 使用自定义客户端，并设置不用于追踪服务
set_default_openai_client(client=client, use_for_tracing=False)  
# 使用兼容的API模式，将默认 API 调用方式设置为 "chat_completions"（聊天补全模式）
set_default_openai_api("chat_completions")  
# 全局禁用 OpenAI 的追踪服务（通常用于第三方兼容接口）
set_tracing_disabled(disabled=True)  

# 定义要使用的模型名称字符串
model_name = 'deepseek-flash'  
# 修改 openai_provider 中的默认模型名称，使其指向上面定义的模型
openai_provider.DEFAULT_MODEL = model_name  
```


# 实例化一个 Agent

```python
# 从 agents 库中导入 Agent（智能体）和 Runner（运行器）类
from agents import Agent, Runner  
# 加载本地设置 # 导入本地的配置模块（加载上一段代码中的环境变量等设置）
import local_settings  

# 实例化一个 Agent 对象，赋值给变量 agent
agent = Agent(  
    # 设置该智能体的名称
    name="AI助手",  
    # 设置智能体的行为指令和系统提示词
    instructions="你是一个友好的AI助手。使用中文进行问答",  
    # 这里一定要指定模型（踩坑吗的）
    model="deepseek-flash",
)

# 使用运行器同步运行该智能体，并传入用户的问题 "你是谁？"，将结果赋值给 result
result = Runner.run_sync(agent, '你是谁？')  
# 打印出运行结果的最终输出内容
print(result.final_output)  
```


# async 语法基础

```python
# 导入 asyncio 库，用于编写异步代码
import asyncio  

# 使用 async 关键字定义一个异步函数 main
async def main():  
    print('123')
    # 使用 await 挂起当前协程 1 秒钟，模拟异步耗时操作（如网络请求）
    await asyncio.sleep(1)  

# 判断当前脚本是否被直接运行（而不是作为模块被导入）
if __name__ == '__main__':  
    # 使用 asyncio.run() 启动事件循环，并运行 main 异步函数
    asyncio.run(main())  
```

# async 结合改造

```python
# 导入 asyncio 库，用于编写异步代码
import asyncio
# 从 agents 库中导入 Agent（智能体）和 Runner（运行器）类
from agents import Agent, Runner

```

# async 改造流式输出

通过 async for 循环监听智能体产生的事件流，并利用 ResponseTextDeltaEvent 捕获每次返回的文本片段（Delta），从而实现类似 ChatGPT 网页版那样逐字打印的“打字机”效果。

做这件事的目的，为了随时能在循环中做出定制化的处理决策，做到精准的控制。

```python
import asyncio  

from agents import Agent, Runner
# 从 openai 库中导入流式文本增量事件类型，用于处理流式输出
from openai.types.responses import ResponseTextDeltaEvent  

import local_settings

agent = Agent(
    name="AI助手",
    instructions="你是一个友好的AI助手。使用中文进行问答",
    model="deepseek-flash",  
)

# 定义一个异步主函数 main
async def main():
    # 以流式（streamed）模式运行智能体，并传入用户提示词
    result = Runner.run_streamed(agent, '讲一个故事至少300个字')  

    # 异步迭代 result 中的流式事件
    async for event in result.stream_events():  
        # 判断事件类型是否为原始响应事件，且数据类型为文本增量事件
        # 注意这个事件类型有很多，后面了解下 比如：RawResponsesStreamEvent 模型触发的事件
        if event.type == "raw_response_event" and isinstance(event.data, ResponseTextDeltaEvent):  
            # 打印出该事件中的文本增量内容（实现打字机效果的流式输出）
            print(event.data.delta)  

if __name__ == '__main__':  
    asyncio.run(main()) 
```
