# 注意

绝对不要直接运行 mcp_server.py。它应该被 main.py（客户端代码）作为子进程静默启动。


# 创建一个 mcp

```python
# 导入操作系统接口模块，用于执行如列出目录等系统操作
import os  
# 从 mcp 库中导入 MCPServer 类，用于快速构建 MCP 服务器
from mcp.server.mcpserver import MCPServer

# 创建一个名为 "MCP Demo" 的 MCPServer 服务器实例
mcp = MCPServer("MCP Demo")  

# 使用装饰器将下面的函数注册为一个 MCP 工具，供客户端调用
@mcp.tool()  
# 定义一个名为 ls 的函数（模拟 Linux 的 ls 命令）
def ls():  
    # 函数的文档字符串，描述了该工具的功能
    """列出目录中的文件名"""  
    # 返回当前目录（"." 表示当前路径）下的所有文件和文件夹名称列表
    return os.listdir(".")  

# 判断当前脚本是否作为主程序直接运行
if __name__ == '__main__':  
    # 启动 MCP 服务器，开始监听和处理请求
    mcp.run()  
```


# 使用

内部核心原理是三步，实际用框架 用 MCPServerStdio 帮你做了
1. 启动MCPServer
2. 从MCPServer获取支持的工具
3. 把工具传递给agent

```python
import asyncio 

from agents import Agent, Runner  
# 从agents.mcp模块导入MCPServerStdio，用于通过标准输入输出启动MCP服务器
from agents.mcp import MCPServerStdio  

import config 

async def main():  

    # 使用async with异步上下文管理器实例化并启动MCP服务器
    # 使用了 async with 语法，这意味着它会自动管理服务器的生命周期；进入代码块时启动服务器进程，退出时关闭
    # 启动后，变量 mcp_server 就可以用来与这个服务器进行交互（比如调用mcp定义的 tools 工具）。
    async with MCPServerStdio(  
            # 设置该MCP服务器的名称为"普通的mcp服务器"
            name="普通的mcp服务器",  
            # 定义启动参数
            params={ 
                # 指定要执行的命令为 python
                "command": "python",  
                # 指定命令的参数，即要运行的脚本文件为 mcp_server.py
                "args": ["mcp_server.py"],  
            }  
    ) as mcp_server:  # 将启动的服务器实例赋值给变量 mcp_server，进入下方代码块

        # 实例化一个Agent（智能体）对象
        agent = Agent(  
            name="AI助手",
            instructions="你是一个友好的AI助手。使用中文进行问答",
            model="deepseek-flash",  
            # 将刚才启动的mcp_server实例作为工具服务器传递给智能体
            mcp_servers=[mcp_server]  
        )  

        result = await Runner.run(agent, '当前目录有几个文件？')  
        print(result.final_output)  

if __name__ == '__main__':  
    # 运行异步主函数main，启动整个程序的执行流程
    asyncio.run(main())  

```
