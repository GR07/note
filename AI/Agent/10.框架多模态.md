# 支持图片对话

核心还是列表，把图片和提示词放入，需要把图片转成 base64 再传输。


```python
import asyncio  
# 导入base64库，用于对二进制数据进行Base64编码
import base64  
from agents import Agent, Runner

import config

agent = Agent(
    name="AI助手",
    instructions="你是一个友好的AI助手。使用中文进行问答",
    model="deepseek-flash",
)

# 定义一个函数，用于将图片文件编码为 Base64 字符串
def encode_image(image_path):  
    # 以二进制只读模式打开指定路径的图片文件
    with open(image_path, "rb") as image_file:  
        # 读取图片的二进制内容
        bin_content = image_file.read()  
        # 将二进制内容进行Base64编码，并解码为UTF-8字符串返回
        return base64.b64encode(bin_content).decode("utf-8")  

# 定义一个异步主函数 main
async def main():
    # 调用上面的函数，读取并编码当前目录下的 "pic.png" 图片
    base64_image = encode_image("pic.png") 

    # 构建发送给智能体的消息列表
    messages = [  
        {
            # 设定消息角色为"用户"
            "role": "user",
            "content": [  # 消息内容是一个列表，包含多种类型的数据
                {
                    # 指定内容类型为输入图片
                    "type": "input_image",  
                    # 使用f-string拼接图片的Data URL格式
                    "image_url": f"data:image/jpeg;base64,{base64_image}",  
                }
            ],  # 图片内容
        },
        {
            # 设定第二条消息的角色也为"用户"
            "role": "user",  
            # 设定具体的文本提问内容
            "content": "图片中是什么内容？",  
        },  # 提示词内容
    ]

    result = await Runner.run(agent, messages)
    print(result.final_output)

if __name__ == '__main__': 
    asyncio.run(main())

```

## 注：不同 agent 的方式有差异化，上面是 openai
